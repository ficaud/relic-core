cmake_minimum_required(VERSION 3.20.0)

option(BUILD_TESTS "Build unit tests with Google Test" OFF)

# ─────────────────────────────────────────────────────────────────────────────
#  Dispatch: native unit tests  vs  Zephyr firmware
# ─────────────────────────────────────────────────────────────────────────────
if(BUILD_TESTS)

  project(relic-core-tests LANGUAGES C CXX)
  enable_testing()
  add_subdirectory(tests)

else()

  # Explicitly list Zephyr modules to avoid the module scanner picking up
  # the self project (root) which contains zephyr/ as a subdirectory.
  set(ZEPHYR_MODULES
      ${CMAKE_CURRENT_SOURCE_DIR}/modules/hal/espressif
      ${CMAKE_CURRENT_SOURCE_DIR}/modules/crypto/mbedtls
      ${CMAKE_CURRENT_SOURCE_DIR}/bootloader/mcuboot
      ${CMAKE_CURRENT_SOURCE_DIR}/modules/lib/picolibc
      ${CMAKE_CURRENT_SOURCE_DIR}/modules/crypto/tf-psa-crypto
      ${CMAKE_CURRENT_SOURCE_DIR}/modules/lib/zcbor
  )

  find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
  # CXX is needed for the ZXing-cpp decode backend (qr_decode_zxing.cpp).
  project(wifi_apsta_mode LANGUAGES C CXX)

  target_sources(app PRIVATE
      src/main.c
      src/access-point/http/src/http_server.c
      src/access-point/http/src/http_handlers.c
      src/access-point/http/src/http_types.c
      src/access-point/http/src/http_router.c
      src/access-point/net/src/dns.c
      src/access-point/net/src/dhcp.c
      src/access-point/wifi/src/wifi_mgr.c
      src/sss/sss.c
      src/compression/encoding/base32.c
      src/compression/encoding/share_base32.c
      src/compression/word_list/wl_codec.c
      src/compression/word_list/bip39_words.c
      src/compression/word_list/slip39_words.c
      src/qrcode/qr_encode.c
      src/svg/qrcode_to_svg.c
  )

  # On-device QR decoding. Only enabled where the board has enough RAM
  # (ESP32-S3). The classic ESP32 (DevKit V1) keeps the decoder out of the
  # firmware entirely and the web page falls back to local jsQR decoding.
  if(CONFIG_RELIC_QR_DECODE_SERVER)
    # Per-board QR decode image limit (Kconfig: RELIC_QR_DECODE_MAX_DIM).
    target_compile_definitions(app PRIVATE
        QR_DECODE_MAX_DIM=${CONFIG_RELIC_QR_DECODE_MAX_DIM}
    )

    if(CONFIG_RELIC_QR_DECODE_BACKEND_ZXING)
      # ZXing-cpp QR-only reader subset (C++20). Compile the individual source
      # files instead of zxing-cpp's own CMake project so the flags stay under
      # our control and no foreign project()/PCH machinery runs.
      set(ZXING_DIR ${CMAKE_CURRENT_SOURCE_DIR}/external/zxing-cpp/core/src)
      target_sources(app PRIVATE
          src/qrcode/qr_decode_zxing.cpp
          ${ZXING_DIR}/Barcode.cpp
          ${ZXING_DIR}/BarcodeFormat.cpp
          ${ZXING_DIR}/BitMatrix.cpp
          ${ZXING_DIR}/CharacterSet.cpp
          ${ZXING_DIR}/CreateBarcode.cpp
          ${ZXING_DIR}/Content.cpp
          ${ZXING_DIR}/ECI.cpp
          ${ZXING_DIR}/Error.cpp
          ${ZXING_DIR}/GTIN.cpp
          ${ZXING_DIR}/JSON.cpp
          ${ZXING_DIR}/ReadBarcode.cpp
          ${ZXING_DIR}/ReedSolomon.cpp
          ${ZXING_DIR}/Utf.cpp
          ${ZXING_DIR}/WriteBarcode.cpp
          ${ZXING_DIR}/ZXingCpp.cpp
          ${ZXING_DIR}/BitArray.cpp
          ${ZXING_DIR}/HRI.cpp
          ${ZXING_DIR}/BinaryBitmap.cpp
          ${ZXING_DIR}/BitSource.cpp
          ${ZXING_DIR}/ConcentricFinder.cpp
          ${ZXING_DIR}/GlobalHistogramBinarizer.cpp
          ${ZXING_DIR}/GridSampler.cpp
          ${ZXING_DIR}/HybridBinarizer.cpp
          ${ZXING_DIR}/LocalGrid.cpp
          ${ZXING_DIR}/MultiFormatReader.cpp
          ${ZXING_DIR}/PerspectiveTransform.cpp
          ${ZXING_DIR}/ResultPoint.cpp
          ${ZXING_DIR}/TextDecoder.cpp
          ${ZXING_DIR}/WhiteRectDetector.cpp
          ${ZXING_DIR}/qrcode/QRCodecMode.cpp
          ${ZXING_DIR}/qrcode/QRErrorCorrectionLevel.cpp
          ${ZXING_DIR}/qrcode/QRVersion.cpp
          ${ZXING_DIR}/qrcode/QRBitMatrixParser.cpp
          ${ZXING_DIR}/qrcode/QRDataBlock.cpp
          ${ZXING_DIR}/qrcode/QRDecoder.cpp
          ${ZXING_DIR}/qrcode/QRDetector.cpp
          ${ZXING_DIR}/qrcode/QRFormatInformation.cpp
          ${ZXING_DIR}/qrcode/QRReader.cpp
          # ECI character-set conversion (used by TextDecoder). C source.
          ${ZXING_DIR}/libzueci/zueci.c
      )
      # PSRAM-backed C++ heap for the decoder (XIAO @ 640px).
      if(CONFIG_RELIC_QR_DECODE_PSRAM)
        target_sources(app PRIVATE src/qrcode/psram_new.cpp)
      endif()
      # src/qrcode/zxing holds the static Version.h (generated by zxing-cpp's
      # CMake upstream; we hard-code the QR-only flags instead).
      target_include_directories(app PRIVATE
          ${ZXING_DIR}
          src/qrcode/zxing
      )
      target_compile_definitions(app PRIVATE ZXING_INTERNAL)
    else()
      # quirc backend (default; kept for comparison/rollback).
      target_sources(app PRIVATE
          src/qrcode/qr_decode.c
          external/quirc/lib/quirc.c
          external/quirc/lib/decode.c
          external/quirc/lib/identify.c
          external/quirc/lib/version_db.c
      )
      target_include_directories(app PRIVATE external/quirc/lib)
      target_compile_definitions(app PRIVATE
          # Shrink quirc's fixed-size buffers for the firmware:
          # - QUIRC_MAX_PAYLOAD: QR shares are at most a few hundred bytes
          #   (the WASM demo links without this and keeps the 8896-byte limit).
          # - QUIRC_MAX_CAPSTONES / QUIRC_MAX_GRIDS: the device decodes a single
          #   QR share per image, so a small finder-pattern pool is enough. This
          #   cuts ~9 KB off quirc's heap footprint, letting a 224x224 image fit.
          QUIRC_MAX_PAYLOAD=2048
          QUIRC_MAX_CAPSTONES=8
          QUIRC_MAX_GRIDS=16
          # The ESP32-S3 FPU is single-precision only; double is emulated in
          # software. quirc's perspective mapping runs per grid cell, so using
          # float cuts the decode from ~2 s down to tens of milliseconds.
          QUIRC_FLOAT_TYPE=float
      )
    endif()
  endif()

  # Auto-generate page_captive.h from assets (HTML + CSS).
  # Generated into the *build* directory (per-board) rather than the shared
  # assets directory: the QR_DECODE_SERVER / QR_MAX_DIM flags baked into the
  # page differ per board, so a shared file would go stale when building
  # several boards in sequence.
  set(ASSETS_DIR    ${CMAKE_CURRENT_SOURCE_DIR}/src/access-point/assets)
  set(ASSETS_OUTPUT ${CMAKE_BINARY_DIR}/page_captive.h)

  if(CONFIG_RELIC_QR_DECODE_SERVER)
    set(QR_DECODE_SERVER_FLAG 1)
  else()
    set(QR_DECODE_SERVER_FLAG 0)
  endif()

  add_custom_command(
      OUTPUT  ${ASSETS_OUTPUT}
      COMMAND ${PYTHON_EXECUTABLE}
              ${CMAKE_CURRENT_SOURCE_DIR}/tools/embed-assets.py
              --qr-max-dim ${CONFIG_RELIC_QR_DECODE_MAX_DIM}
              --qr-decode-server ${QR_DECODE_SERVER_FLAG}
              ${ASSETS_DIR}
              ${ASSETS_OUTPUT}
      DEPENDS ${ASSETS_DIR}/index.html
              ${ASSETS_DIR}/split.html
              ${ASSETS_DIR}/unsplit.html
              ${ASSETS_DIR}/style.css
              ${ASSETS_DIR}/scripts/split.js
              ${ASSETS_DIR}/scripts/unsplit.js
              ${ASSETS_DIR}/scripts/jsQR.js
              ${CMAKE_CURRENT_SOURCE_DIR}/tools/embed-assets.py
      COMMENT "Generating page_captive.h from assets..."
  )

  target_sources(app PRIVATE ${ASSETS_OUTPUT})

  target_include_directories(app PRIVATE
      ${CMAKE_BINARY_DIR}
      src/access-point/http/inc
      src/access-point/net/inc
      src/access-point/wifi/inc
      src/access-point/assets
      src/sss
      src/compression
      src/compression/encoding
      src/compression/word_list
      src/qrcode
      src/svg
  )

endif()
